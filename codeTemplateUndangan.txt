document.addEventListener('DOMContentLoaded', () => {

    // --- Splash Screen & Audio ---
    const splashScreen = document.getElementById('splash-screen');
    const mainContent = document.getElementById('main-content');
    const openBtn = document.getElementById('open-invitation');
    const audio = document.getElementById('bg-music');
    const audioToggle = document.getElementById('audio-toggle');
    let isPlaying = false;

    // Retrieve guest name from URL if any (e.g. ?kepada=Tamu+Spesial)
    const urlParams = new URLSearchParams(window.location.search);
    const namaTamu = urlParams.get('kepada');

    if (namaTamu) {
        document.getElementById('nama-tamu').innerText = namaTamu;
    }

    openBtn.addEventListener('click', () => {
        // Trigger gate animation
        splashScreen.classList.add('open');

        // Show main content
        mainContent.classList.remove('hidden');

        // Play audio
        audio.play().then(() => {
            isPlaying = true;
            updateAudioIcon();
        }).catch(err => {
            console.log("Audio autoplay prevented by browser", err);
        });

        // Hide splash screen after animation
        setTimeout(() => {
            splashScreen.classList.add('fade-out');
        }, 1500); // 1.5s matches CSS transition
    });

    audioToggle.addEventListener('click', () => {
        if (isPlaying) {
            audio.pause();
        } else {
            audio.play();
        }
        isPlaying = !isPlaying;
        updateAudioIcon();
    });

    function updateAudioIcon() {
        const icon = audioToggle.querySelector('i');
        if (isPlaying) {
            icon.classList.remove('fa-music');
            icon.classList.remove('fa-volume-xmark');
            icon.classList.add('fa-volume-high');
            audioToggle.style.animation = 'pulse 2s infinite';
        } else {
            icon.classList.remove('fa-volume-high');
            icon.classList.add('fa-volume-xmark');
            audioToggle.style.animation = 'none';
        }
    }

    // --- Scroll Animations (Intersection Observer) ---
    const observerOptions = {
        root: null,
        rootMargin: '0px',
        threshold: 0.15
    };

    const observer = new IntersectionObserver((entries) => {
        entries.forEach(entry => {
            if (entry.isIntersecting) {
                entry.target.classList.add('visible');
            } else {
                // Remove the class when out of view so it can animate again
                entry.target.classList.remove('visible');
            }
        });
    }, observerOptions);

    document.querySelectorAll('.animate-fade-up, .animate-slide-left, .animate-slide-right').forEach(el => {
        observer.observe(el);
    });

    // --- Enhanced Parallax Effect ---
    const parallaxSections = document.querySelectorAll('.parallax-bg');
    window.addEventListener('scroll', () => {
        let scrollPosition = window.scrollY;
        
        parallaxSections.forEach(section => {
            // [BUG FIX] Menonaktifkan pergeseran backgroundPositionY karena menyebabkan foto
            // mengulang (tiling) / terpotong jika foto tidak cukup tinggi untuk menutupi pergeseran.
            // section.style.backgroundPositionY = (scrollPosition * 0.5) + 'px';
            
            // If it's the hero section, also animate the content slightly for depth
            if (section.id === 'hero') {
                const heroContent = section.querySelector('.hero-content');
                if (heroContent) {
                    heroContent.style.transform = `translateY(${scrollPosition * 0.3}px)`;
                    // Optional: Fade out content slightly as user scrolls down
                    heroContent.style.opacity = Math.max(1 - (scrollPosition / 600), 0);
                }
            }
        });
    });

    // --- Countdown Timer ---
    // Gunakan date string ISO atau parse format dari template variabel
    // Contoh format: "Sabtu, 12 Desember 2026" - JS Date mungkin gagal nge-parse string ini, 
    // namun kita bisa menggunakan pendekatan sederhana: menggunakan tanggal format "YYYY-MM-DD" jika ada
    // Atau fallback ke tanggal hardcoded untuk demo
    const targetDateStr = "{{weddingDate}} {{akadTime}}".replace('WITA', '').replace('- Selesai', '').trim(); 
    // Catatan: Jika format weddingDate bukan standar (misal "Sabtu, 12 Desember 2026"), new Date() mungkin return Invalid Date.
    // Untuk simplifikasi demo, kita gunakan date sederhana:
    let countDownDate = new Date(targetDateStr).getTime();
    
    if (isNaN(countDownDate)) {
       countDownDate = new Date("Dec 12, 2026 09:00:00").getTime(); // Fallback aman
    }

    const countdownTimer = setInterval(() => {
        const now = new Date().getTime();
        const distance = countDownDate - now;

        const days = Math.floor(distance / (1000 * 60 * 60 * 24));
        const hours = Math.floor((distance % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
        const minutes = Math.floor((distance % (1000 * 60 * 60)) / (1000 * 60));
        const seconds = Math.floor((distance % (1000 * 60)) / 1000);

        // Update HTML
        document.getElementById("days").innerHTML = days.toString().padStart(2, '0');
        document.getElementById("hours").innerHTML = hours.toString().padStart(2, '0');
        document.getElementById("minutes").innerHTML = minutes.toString().padStart(2, '0');
        document.getElementById("seconds").innerHTML = seconds.toString().padStart(2, '0');

        // If the countdown is finished
        if (distance < 0) {
            clearInterval(countdownTimer);
            document.querySelector(".countdown").innerHTML = "<p class='font-lato text-gold'>Acara sedang berlangsung / telah selesai.</p>";
        }
    }, 1000);



    // --- Copy to Clipboard ---
    const copyBtns = document.querySelectorAll('.copy-btn');
    copyBtns.forEach(btn => {
        btn.addEventListener('click', function () {
            const targetId = this.getAttribute('data-target');
            const targetText = document.getElementById(targetId).innerText;

            navigator.clipboard.writeText(targetText).then(() => {
                const originalText = this.innerText;
                this.innerText = 'Copied!';
                setTimeout(() => {
                    this.innerText = originalText;
                }, 2000);
            }).catch(err => {
                console.error('Failed to copy: ', err);
            });
        });
    });

    // --- RSVP Form Handling ---
    const rsvpForm = document.getElementById('rsvp-form');
    rsvpForm.addEventListener('submit', (e) => {
        e.preventDefault();
        const btn = rsvpForm.querySelector('button');
        const originalText = btn.innerText;
        btn.innerText = "Mengirim...";
        btn.disabled = true;

        // Simulate API call
        setTimeout(() => {
            btn.innerText = "Terkirim!";
            btn.style.backgroundColor = "var(--c-green)";
            rsvpForm.reset();

            setTimeout(() => {
                btn.innerText = originalText;
                btn.style.backgroundColor = "var(--c-gold)";
                btn.disabled = false;
            }, 3000);
        }, 1500);
    });

});
